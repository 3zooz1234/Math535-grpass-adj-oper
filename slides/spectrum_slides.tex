  \subsection{Revision}
  \begin{frame}{\subsecname}
    \begin{defn}
    Let $T$ be a linear operator, $\lambda \in \mathbb{C}$ is called an eigenvalue of $T$, if there exists $x \in H$ with $x\not=0$ such that

    $$
    Tx=\lambda x
    $$
    \end{defn}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{exm}
      Consider
      $$
      \begin{aligned}
        T:\mathbb{R}^{2}&\longrightarrow\mathbb{R}^{2} \\
        x&\longmapsto\begin{pmatrix}1&1\\0&1\\\end{pmatrix}x
      \end{aligned}
      $$
      $1\in\mathbb{C}$ is an eigenvalue of $T$, since there is $(1,0)\in\mathbb{R}^{2}$ such that
      $$
      T\begin{pmatrix}1\\0\\\end{pmatrix} = \begin{pmatrix}1&1\\0&1\\\end{pmatrix}\begin{pmatrix}1\\0\\\end{pmatrix} = 1\cdot\begin{pmatrix}1\\0\\\end{pmatrix}
      $$
    \end{exm}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{thm} \label{Inv}
      Let $\mathcal{H}$ be a Hilbert space and $T: \mathcal{H}\rightarrow\mathcal{H}$ be bounded linear then $T$ is invertible $\Longleftrightarrow$  $\operatorname{ker}T^{*}=\{0\}$ and $\exists \alpha>0$ such that $\|T(x)\| \geqslant \alpha\|x\| \quad \forall x \in\mathcal{H}$.
    \end{thm}
    \pause
    \begin{cor} \label{CorInv}
    Let $\mathcal{H}$ be a Hilbert space and $T: \mathcal{H}\rightarrow\mathcal{H}$ be bounded linear then $T$ is invertible $\Longleftrightarrow$ $\exists \beta>0$ such that $\|T^{*}(x)\| \geqslant \beta\|x\|$ and $\exists \alpha>0$ such that $\|T(x)\| \geqslant \alpha\|x\| \quad \forall x \in\mathcal{H}$.
    \end{cor}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
      If $T$ is invertible, then so is $T^{*}$. So, by Theorem \ref{Inv}, $\exists \beta>0$ such that $\|T^{*}(x)\| \geqslant \beta\|x\|$ and $\exists \alpha>0$ such that $\|T(x)\| \geqslant \alpha\|x\|$ for all $x\in H$. Conversely, if $\exists \beta>0$ such that $\|T^{*}(x)\| \geqslant \beta\|x\|$, then if $T^{*}u=0$ for some $u\in H$, then
      $$
      \begin{aligned}
      0 = \lVert T^{*}u\rVert&\geq\beta\lVert u\rVert \\
                             &\implies u=0
      \end{aligned}
      $$
      So, $\ker T^{*}=\{0\}$. By Theorem \ref{Inv}, $T$ is invertible.
    \end{proof}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{defn}
      A linear operator $T:\mathcal{H}\to\mathcal{H}$ is said to be \textbf{normal} if $TT^{*}=T^{*}T$.
    \end{defn}
    \begin{defn}
      A linear operator $T:\mathcal{H}\to\mathcal{H}$ is said to be \textbf{self-adjoint} if $T=T^{*}$.
    \end{defn}
    \begin{defn}
      A linear operator $U:\mathcal{H}\to\mathcal{H}$ is said to be \textbf{uniatry} if $UU^{*}=U^{*}U=I$
    \end{defn}
  \end{frame}

  \subsection{Definition of Spectrum}
  \begin{frame}{\subsecname}
    \begin{defn}
      Let $T$ be a linear operator. The resolvent set of $T$, denoted by $\rho(T)$, is $\rho(T) = \{\lambda\in\mathbb{C}: T-\lambda I\text{ is invertible}\}$. The spectrum of $T$, denoted by $\sigma(T)$ is $\sigma(T)=\mathbb{C}\backslash\rho(T)$, or equivalently, $\sigma(T) = \{\lambda\in\mathbb{C}: T-\lambda I\text{ is not invertible}\}$
    \end{defn}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{rem} \label{defRho}
      Notice that, if $T\in B(\mathcal{H})$, by Theorem \ref{Inv}, we can rewrite the resolvent set of $T$ as
      $$
      \begin{aligned}
        \rho(T) = \{\lambda\in\mathbb{C}&: \operatorname{ker}(T-\lambda I)^{*}=\{0\}\\
                                        &\text{ and }\lVert (T-\lambda I)x\rVert\geq\alpha\lVert x\rVert\text{ for some }\alpha>0\}
      \end{aligned}
      $$
      Alternatively, we can rewrite the resolvent set of $T$ as
      $$
      \begin{aligned}
        \rho(T) = \{\lambda\in\mathbb{C}&: \lVert (T-\lambda I)^{*}x\rVert\geq\beta\lVert x\rVert\\
                                        &\text{ and }\lVert (T-\lambda I)x\rVert\geq\alpha\lVert x\rVert\text{ for some }\beta,\alpha>0\}
      \end{aligned}
      $$
      by Corollary \ref{CorInv}
    \end{rem}
  \end{frame}

  \subsection{Relationship between Eigenvalue and Spectrum}
  \begin{frame}{\subsecname}
    \begin{thm}
    If $\lambda$ is an eigenvalue of $T$ , then $\lambda \in \sigma(T)$
    \end{thm}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
      Suppose that $\lambda\notin\sigma(T)$, then $T-\lambda I$ is invertible. So, we have $(T-\lambda I)u=0\implies u = (T-\lambda I)^{-1}(T-\lambda I)u = (T-\lambda I)^{-1}(0)=0$. So, if there is $u\in\mathcal{H}$ such that $Tu=\lambda u$, then $u=0$. So, $\lambda$ is not an eignvalue of $T$.
    \end{proof}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{thm}
      Let $\mathcal{H}$ be a finite dimensional Hilbert space, and let $T:\mathcal{H}\to\mathcal{H}$ be linear. If $\lambda\in\sigma(T)$, then $\lambda$ is an eigenvalue of $T$.
    \end{thm}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{rem}
    For finite dimesonial Hilbert space, the spectrum of $T$ consist only the eigenvalues of $T$. However, in infinite dimesional Hilbert space, we may have no eigenvalues. For instance, consider the shift operator. Here, it has no eigenvalues, since if we consider $(T-\lambda I)x=0$, we see that
    $$
    \begin{aligned}
      (-\lambda x_{1}, x_{1}-\lambda x_{2}, x_{2}-\lambda x_{3},\cdots)&=(0,0,0,\cdots)\\
                                                                       &\implies x = (0,0,0,\cdots)
    \end{aligned}
    $$
    So, $T$ does not have an eigenvalue. However, we showed that $T$ is not invertible. So, $T-0I$ is not invertible and $0\in\sigma(T)$. So, the spectrum of an operator does not only consist of eigenvalues.
    \end{rem}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
      Suppose that $\lambda$ is not an eigenvalue of $T$, then there is nonzero $u\in \mathcal{H}$ such that $Tu=\lambda u$. So, there is no nonzero $u\in\mathcal{H}$ such that $(T-\lambda I)u=0$. So, $\ker(T-\lambda I)=\{0\}$. By rank-nullity theorem, we have
      $$
      \begin{aligned}
        \dim\mathcal{H} &= \dim\ker(T-\lambda I) + \dim\im(T-\lambda I) \\
               &= 0 + \dim\im(T-\lambda I) \\
               &= \dim\im(T-\lambda I)
      \end{aligned}
      $$
      So, $\im(T-\lambda I)=\mathcal{H}$. This means that $(T-\lambda I)$ is surjective, and therefore, invertible. So, $\lambda\in\rho(T)$, and therefore, $\lambda\notin\sigma(T)$.
    \end{proof}
  \end{frame}

  \subsection{Properties and Theorems}
  \begin{frame}{\subsecname}
    \begin{prop}
      Let $\mathcal{H}$ be a Hilbert space. Let $T:\mathcal{H}\to\mathcal{H}$ be linear operator. The following properties hold:
      \begin{enumerate}
        \item $\sigma(I) = \{1\}$
        \pause
        \item $\sigma\left(T^{*}\right)=\{\bar{\lambda}: \lambda \in \sigma(T)\}$
        \pause
        \item If $T$ is invertible, then $\sigma\left(T^{-1}\right)=\left\{\mu^{-1}: \mu \in \sigma(T)\right\}$
        \pause
        \item If $T$ is self-adjoint, then $\sigma(T)\subseteq\mathbb{R}$.
        \pause
        \item If $U:\mathcal{H}\to\mathcal{H}$ is invertible, then $\sigma(UTU^{-1})=\sigma(T)$
      \end{enumerate}
    \end{prop}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
      \begin{itemize}
        \item\textbf{Proof of 1:} Suppose that $\lambda\in\sigma(I)$, then $I-\lambda I=(1-\lambda)I$ is not invertible. $(1-\lambda)I$ is not invertible if and only if $1-\lambda=0$, since $1-\lambda\in\mathbb{F}$. So, this is true if and only if $\lambda=1$.
          \pause

        \item\textbf{Proof of 2:} Suppose that $\mu\in\sigma(T^{*})$, then $T^{*}-\mu I$ is not invertible. Observe that $T^{*}-\mu I$ is invertible if and only if $(T^{*}-\mu I)^{*}=T-\bar{\mu}I$ is invertible, since if $(T^{*}-\mu I)^{*}$ is invertible, then $((T^{*}-\mu I)^{*})^{*}=T^{*}-\mu I$ is invertible. So, $T^{*}-\mu I$ is not invertible iff $(T^{*}-\mu I)^{*}=T-\bar{\mu}I$ is not invertible. So, $\bar{\mu}\in\sigma(T)$, i.e. $\mu\in\{\bar{\lambda}: \lambda\in\sigma(T)\}$.
        \end{itemize}
      \end{proof}
    \end{frame}

    \begin{frame}{\subsecname}
      \begin{proof}
        \begin{itemize}
          \pause
        \item\textbf{Proof of 3:} If $T$ is invertible, i.e. $T-0I$ is invertible, then $0\notin\sigma(T)$. Similarly, $0\notin\sigma(T^{-1})$. Suppose that $\mu\in\sigma(T^{-1})$, then $T^{-1}-\mu I$ is not invertible. Observe that $T^{-1}-\mu I$ is invertible if and only if $T(T^{-1}-\mu I)=I-\mu T$ is invertible, since if $T(T^{-1}-\mu I)$ has an inverse, $K$, then $KT$ would be an inverse of $T^{-1}-\mu I$. So, $T^{-1}-\mu I$ is not invertible iff $T(T^{-1}-\mu I)=I-\mu T$ is not invertible. Also, $I-\mu T$ is not invertible iff $T-\mu^{-1}I$, where $\mu^{-1}$ exists, since $\mu\not=0$. So, $\mu^{-1}\in\sigma(T)$, i.e. $\mu\in\{\lambda^{-1}: \lambda\in\sigma(T)\}$.
          \pause

        \item\textbf{Proof of 5:} Suppose that $\lambda\in\sigma(UTU^{-1})$, then $UTU^{-1}-\lambda I$ is not invertible. Observe that $UTU^{-1}-\lambda I=U(T-\lambda)U^{-1}$ is not invertible iff $T-\lambda I$ is not invertible. So, $\lambda\in\sigma(T)$.
      \end{itemize}
    \end{proof}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{exm}{Quantum Physics}
    We have seen that for an observable $t$, we associate a self-adjoint operator $T$.\textbf{ The eigenvalues of the operator $T$ represent the possible values of that observable $t$.}
    \pause
    For example, if we want to find the possible values of the energy, we need to find the eigenvalues of $H$. In other words, we solve
    \[
    H\psi=E\psi=\left(-\frac{\hbar^2}{2m}\frac{\partial^2}{\partial x^2}+V(x)\right)\psi.
    \]
    \pause
    This is the famous \textbf{Schrödinger} equation.
    \end{exm}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{thm} \label{SpectIsBd}
    Let $\mathcal{H}$ be a complex Hilbert space and let $T \in B(\mathcal{H})$. If $|\lambda|>\|T\|$ then $\lambda \notin \sigma(T)$.
    \end{thm}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
      Let $u\in\operatorname{ker}(T-\lambda I)^{*}$, then $(T-\lambda I)^{*}u = (T^{*}-\bar{\lambda}I)u = 0$. So, $T^{*}u-\bar{\lambda}u = 0\implies T^{*}u = \bar{\lambda}u$. So,
      $$
      \begin{aligned}
        |\lambda|\lVert u\rVert = |\bar{\lambda}|\lVert u\rVert = \lVert\bar{\lambda}u\rVert = \lVert T^{*}u\rVert&\leq\lVert T^{*}\rVert\lVert u\rVert = \lVert T\rVert\lVert u\rVert\\\\
                                                                                                                  &\implies (|\lambda|-\lVert T\rVert)\lVert u\rVert\leq 0
      \end{aligned}
      $$
      Since $|\lambda|>\lVert T\rVert$, then $\lVert u\rVert\leq 0\implies \lVert u\rVert = 0\implies u=0$. Hence, $\operatorname{ker}(T-\lambda I)^{*} = \{0\}$.
    \end{proof}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
      We also have that
      $$
      \begin{aligned}
        \lVert (T-\lambda I)x\rVert = \rVert Tx-\lambda x\lVert&\geq|\lVert Tx\rVert - |\lambda|\lVert x\rVert|\\ &= \Big|\frac{\lVert Tx\rVert}{\lVert x\rVert} - |\lambda|\Big|\lVert x\rVert\\
                                                                                                               &= \Big(|\lambda|-\frac{\lVert Tx\rVert}{\lVert x\rVert}\Big)\lVert x\rVert\\
                                                                                                               &\geq (|\lambda|-\lVert T\lVert)\lVert x\rVert\\
                                                                                                               &= \alpha\lVert x\rVert
      \end{aligned}
      $$
      where $\alpha:=|\lambda|-\lVert T\rVert$ is positive, since $|\lambda|>\lVert T\rVert$. So, $\lambda\in\rho(T)$, by first equality in Remark \ref{defRho}. Hence, $\lambda\notin\sigma(T)$.
    \end{proof}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{cor} \label{SpectIsBd2}
      If $T\in B(\mathcal{H})$, then $\sigma(T)$ is bounded.
    \end{cor}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
      If $\lambda\in\sigma(T)$, then, by the contrapositive of Theorem \ref{SpectIsBd}, $|\lambda|\leq\lVert T\rVert$. So, $\lambda\in\overline{B(0,\lVert T\rVert)}$. Hence, $\sigma(T)\subseteq\overline{B(0,\lVert T\rVert)}$, i.e. $\sigma(T)$ is bounded.
    \end{proof}
  \end{frame}


  \begin{frame}{\subsecname}
    \begin{cor}
    If $\mathcal{H}$ is a complex Hilbert space and $U \in B(\mathcal{H})$ is unitary then

    $$
    \sigma(U) \subseteq\{\lambda \in \mathbb{C}:|\lambda|=1\}
    $$
    \end{cor}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{prop}
    If $\mathcal{H}$ is a complex Hilbert space and $U \in B(\mathcal{H})$ is unitary then
    $$
    \sigma(U) \subseteq\{\lambda \in \mathbb{C}:|\lambda|=1\}
    $$
    \end{prop}
    \begin{proof}
    If $|\lambda|>1=\lVert U\rVert$, then $\lambda\notin\sigma(U)$. Moreover, if $|\lambda|<1$, then
    $$
    U-\lambda I = -\lambda U(U^{-1}-\frac{1}{\lambda}I)
    $$
    is invertible, since $-\lambda U$ is invertible and $\Big|\frac{1}{\lambda}\Big|>1=\lVert U^{-1}\rVert$. So, $\lambda\notin\sigma(U)$. Therefore, $|\lambda|=1$ if $\lambda\in\sigma(U)$.
    \end{proof}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{thm} \label{SpectIsCl}
      If $T\in B(\mathcal{H})$, then $\sigma(T)\subseteq\mathbb{C}$ is a closed set.
    \end{thm}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
    We will show that $\rho(T)=\mathbb{C}\backslash\sigma(T)$ is open. Let $\lambda\in\rho(T)$. This means that, by second equality in Remark \ref{defRho}, there is $\alpha>0$ such that
    $$
    \lVert(T-\lambda I)u\rVert\geq\alpha\lVert u\rVert\implies\frac{\lVert(T-\lambda I)u\rVert}{\lVert u\rVert}\geq\alpha
    $$
    Since the quantity $\frac{\lVert(T-\lambda I)u\rVert}{\lVert u\rVert}$ has a lower bound, its infimum exists. Similarly, by second equality in Remark \ref{defRho}, the infimum of the quantity $\frac{\lVert(T-\lambda I)^{*}u\rVert}{\lVert u\rVert}$ exists.
  \end{proof}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
    Now, let
    $$
    \begin{aligned}
      \gamma&:=\inf_{u\in H\backslash\{0\}}\frac{\lVert(T-\lambda I)u\rVert}{\lVert u\rVert} \quad
      \gamma'&:=\inf_{u\in G\backslash\{0\}}\frac{\lVert(T-\lambda I)^{*}u\rVert}{\lVert u\rVert}
    \end{aligned}
    $$
    and let $\delta:=\frac{1}{2}\min(\gamma,\gamma')$. We want to show that
    $$
    B(\lambda,\delta)\subseteq\rho(T)
    $$
    \end{proof}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
      Let $\tilde{\lambda}\in B(\lambda,\delta)$, then $|\tilde{\lambda}-\lambda|<\delta$. First, we observe that
      $$
      \begin{aligned}
        \lVert(T-\tilde{\lambda}I)u\rVert&=\lVert(T-\lambda I)u-(\tilde{\lambda}-\lambda)I(u)\rVert\\&\geq|\lVert(T-\lambda I)u\rVert-|\bar{\lambda}-\lambda|\lVert u\rVert| \\
                                                                                                  &=\Big|\frac{\lVert(T-\lambda I)u\rVert}{\lVert u\rVert}-|\tilde{\lambda}-\lambda|\Big|\lVert u\rVert \\
                                                                                                  &=\Big(\frac{\lVert(T-\lambda I)u\rVert}{\lVert u\rVert}-|\tilde{\lambda}-\lambda|\Big)\lVert u\rVert \\
                                                                                                  &>(\gamma-\frac{1}{2}\min(\gamma,\gamma'))\lVert u\rVert \\
                                                                                                  &\geq(\gamma-\frac{\gamma}{2})\lVert u\rVert \\
                                                                                                  &=\frac{\gamma}{2}\lVert u\rVert
      \end{aligned}
      $$
    \end{proof}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{proof}
      Similarly, we have
      $$
      \begin{aligned}
        \lVert(T-\tilde{\lambda}I)^{*}u\rVert&=\lVert(T-\lambda I)^{*}u-\overline{(\tilde{\lambda}-\lambda)}I(u)\rVert\\&\geq|\lVert(T-\lambda I)^{*}u\rVert-|\overline{\bar{\lambda}-\lambda}|\lVert u\rVert| \\
                                                                                                                     &=|\lVert(T-\lambda I)^{*}u\rVert-|\tilde{\lambda}-\lambda|\lVert u\rVert| \\
                                                                                                                     &=\Big|\frac{\lVert(T-\lambda I)^{*}u\rVert}{\lVert u\rVert}-|\tilde{\lambda}-\lambda|\Big|\lVert u\rVert \\
                                                                                                                     &=\Big(\frac{\lVert(T-\lambda I)^{*}u\rVert}{\lVert u\rVert}-|\tilde{\lambda}-\lambda|\Big)\lVert u\rVert \\
                                                                                                                     &>(\gamma'-\frac{1}{2}\min(\gamma,\gamma'))\lVert u\rVert 
                                                                                                                     \geq(\gamma'-\frac{\gamma'}{2})\lVert u\rVert 
                                                                                                                     =\frac{\gamma'}{2}\lVert u\rVert
      \end{aligned}
      $$
      So, by the second equality in Remark \ref{Inv}, $\tilde{\lambda}\in\rho(T)$. Therefore, $B(\lambda,\delta)\subseteq\rho(T)$.
    \end{proof}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{cor}
      If $T\in B(\mathcal{H})$, then $\sigma(T)$ is compact in $\mathbb{C}$
    \end{cor}
  \end{frame}

  \subsection{The Shift Operator}
  \begin{frame}{\subsecname}
    \begin{exm}
      We would like to find the spectrum of the unilateral shift, $T:\ell^{2}\to\ell^{2}$, defined by
      $$
      T(x_{1},x_{2},\cdots) = (0,x_{1},x_{2},\cdots)
      $$
      Since $T$ has been shown to have no eigenvalue, we will consider the eigenvalues of $T^{*}$ instead and use the properties of the spectrum to have a lower bound for the spectrum.
    \end{exm}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{exm}
    %We claim that if $|\lambda|$
      Suppose that $\lambda\in\mathbb{C}$ such that $|\lambda|<1$, then consider the sequence
      \pause
      $$
      x = (1,\lambda,\lambda^{2},\cdots)
      $$
      \pause
      \textbf{Claim: }$x\in\ell^{2}$
      \pause
      \newline
      We have
      $$
      \begin{aligned}
        \sum^{\infty}_{n=0}|x_{n}|^{2} = \sum^{\infty}_{n=0}(|\lambda|^{n})^{2} = \sum^{\infty}_{n=0}(|\lambda|^{2})^{n} = \frac{1}{1-|\lambda|^{2}} < \infty
      \end{aligned}
      $$
    \end{exm}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{exm}
      Now, observe that
      \pause
      $$
      \begin{aligned}
        T^{*}x &= T^{*}(1,\lambda,\lambda^{2},\cdots) \\
           &= (\lambda,\lambda^{2},\lambda^{3},\cdots) = \lambda(1,\lambda,\lambda^{2},\cdots) = \lambda x
      \end{aligned}
      $$
      \pause
      So, every $\lambda\in\mathbb{C}$ such that $|\lambda|<1$ is an eigenvalue of $T^{*}$.
      \pause
      Now, we have
      $$
      \{\lambda:|\lambda|<1\}=\{\bar{\lambda}:|\lambda|<1\}\subseteq\sigma(T)
      $$
    \end{exm}
  \end{frame}

  \begin{frame}{\subsecname}
    \begin{exm}
      So,
      $$
      \{\lambda\in\mathbb{C}:|\lambda|<1\}\subseteq\sigma(T)\subseteq\{\lambda\in\mathbb{C}:|\lambda|\leq 1\}
      $$
      \pause
      or
      $$
      B(0,1)\subseteq\sigma(T)\subseteq\overline{B(0,1)}
      $$
      \pause
      Since $\sigma(T)$ is closed, we have
      $$
      \sigma(T)=\overline{B(0,1)}=\{\lambda\in\mathbb{C}:|\lambda|\leq 1\}
      $$
    \end{exm}
  \end{frame}
